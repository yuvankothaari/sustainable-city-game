<script>
let budget, eco, env, happy, gameActive = false;

const projects = [
  {name:"☀️", cost:30, eco:10, env:25, happy:10},
  {name:"🚇", cost:40, eco:20, env:20, happy:25},
  {name:"🏭", cost:25, eco:30, env:-20, happy:5},
  {name:"🌳", cost:10, eco:5, env:20, happy:20},
  {name:"♻️", cost:15, eco:10, env:25, happy:15},
  {name:"🛣️", cost:20, eco:15, env:-10, happy:10}
];

// Chart setup
let chart = new Chart(document.getElementById("chart"), {
  type: 'bar',
  data: {
    labels: ['Economy', 'Environment', 'Happiness'],
    datasets: [{
      data: [0,0,0]
    }]
  }
});

function updateChart(){
  chart.data.datasets[0].data = [eco, env, happy];
  chart.update();
}

// Create project buttons
const container = document.getElementById("projects");
projects.forEach((p,i)=>{
  let div = document.createElement("div");
  div.className = "card";
  div.innerHTML = `
    <h3>${p.name}</h3>
    <p>₹${p.cost} Cr</p>
    <button onclick="build(${i})">Build</button>
  `;
  container.appendChild(div);
});

// Map
const map = document.getElementById("map");
for(let i=0;i<16;i++){
  let cell = document.createElement("div");
  cell.className="cell";
  map.appendChild(cell);
}

function build(i){
  if(!gameActive) return alert("Start the game first!");
  let p = projects[i];

  if(budget >= p.cost){
    budget -= p.cost;
    eco += p.eco;
    env += p.env;
    happy += p.happy;

    document.getElementById("budget").innerText = budget;
    updateChart();

    // add to map
    let empty = [...document.querySelectorAll(".cell")].find(c=>c.innerHTML=="");
    if(empty) empty.innerHTML = p.name;
  }
}

// Start Game
function startGame(){
  budget = 100;
  eco = 0;
  env = 0;
  happy = 0;
  gameActive = true;
  document.getElementById("budget").innerText = budget;
  document.getElementById("result").innerText = "";

  document.querySelectorAll(".cell").forEach(c=>c.innerHTML="");
  updateChart();
}

// End Game
function finishGame(){
  if(!gameActive) return;

  // Random Event
  let events = [
    ()=>{ if(env<30) happy-=20; return "🌪️ Pollution Crisis"; },
    ()=>{ if(eco>50) eco+=20; return "💼 Economic Boom"; },
    ()=>{ if(env>50) happy+=15; return "🌳 Climate Award"; }
  ];

  let event = events[Math.floor(Math.random()*events.length)]();

  let total = eco + env + happy;
  let title = total>120?"🌍 Champion": total>90?"⚖️ Balanced":"⚠️ Poor";

  document.getElementById("result").innerText =
    event + " | Score: " + total + " | " + title;

  saveScore(total);
  showLeaderboard();
  gameActive = false;
}

// Leaderboard
function saveScore(score){
  let scores = JSON.parse(localStorage.getItem("scores") || "[]");
  scores.push(score);
  localStorage.setItem("scores", JSON.stringify(scores));
}

function showLeaderboard(){
  let scores = JSON.parse(localStorage.getItem("scores") || "[]");
  scores.sort((a,b)=>b-a);

  let html = scores.slice(0,5).map(s=>"<p>"+s+"</p>").join("");
  document.getElementById("leaderboard").innerHTML = html;
}

showLeaderboard();
